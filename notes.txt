//branching is nothing but pointers to commit 
git branch name

git branch -d name     --delete

git push origin --delete name   --delete remote branch 

git merge name     to merge

git merge name -m "message"         --merge with a message



///stash temp stores works making working tree clean for new commits
git stash 
git stash push -m "wip on search feature"           ---stash with a message
git stash list   
git stash pop                       --applies but pop the stash 
git stash apply                      --applies changes and keeps in the list 
git stash apply stash_id              --apply specific stash
git stash clear                       --clears all stash
git stash drop stash_id                --delete specific stash
